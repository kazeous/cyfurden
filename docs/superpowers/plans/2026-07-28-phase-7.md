```ts
export const DEFAULT_GACHA_RATES = Object.freeze({
  common: 70,
  rare: 24,
  epic: 5,
  legendary: 1,
});

type RateRecord = Record<keyof typeof DEFAULT_GACHA_RATES, number>;

const KEY_ALIASES: Record<string, keyof typeof DEFAULT_GACHA_RATES> = {
  common: "common",
  COMMON: "common",
  rare: "rare",
  RARE: "rare",
  epic: "epic",
  EPIC: "epic",
  legendary: "legendary",
  LEGENDARY: "legendary",
};

export function readGachaRates(value: unknown): {
  value: RateRecord;
  usedLegacyKeys: boolean;
} {
  const normalized: Record<string, number> = {};
  let usedLegacyKeys = false;

  if (value && typeof value === "object") {
    for (const [rawKey, rawValue] of Object.entries(value as Record<string, unknown>)) {
      const canonical = KEY_ALIASES[rawKey];
      if (!canonical) continue;
      if (typeof rawValue !== "number" || !Number.isFinite(rawValue)) continue;
      normalized[canonical] = rawValue;
      if (rawKey !== canonical) usedLegacyKeys = true;
    }
  }

  return {
    value: { ...DEFAULT_GACHA_RATES, ...normalized } as RateRecord,
    usedLegacyKeys,
  };
}
```

```ts
export const HO_CHI_MINH_OFFSET_MINUTES = 7 * 60;
export const TODAY_QUERY = "today=1";

export function hoChiMinhTodayWindow(reference: Date): {
  start: Date;
  end: Date;
} {
  const utcMillis = reference.getTime();
  const startMillis =
    Math.floor(
      (utcMillis - HO_CHI_MINH_OFFSET_MINUTES * 60 * 1000) /
        (24 * 60 * 60 * 1000),
    ) *
      (24 * 60 * 60 * 1000) +
    HO_CHI_MINH_OFFSET_MINUTES * 60 * 1000;
  const endMillis = startMillis + 24 * 60 * 60 * 1000;
  return { start: new Date(startMillis), end: new Date(endMillis) };
}

export function parseTodayQuery(value: string | undefined | null): boolean {
  if (!value) return false;
  return value === TODAY_QUERY || value === "1";
}
```

```ts
type PaymentContext = {
  bankName: string;
  accountName: string;
  accountNumber: string;
};

type IdentityContext = {
  name: string;
  description: string;
};

type EventContext = {
  location: string;
  hours: string;
};

export type BoothChecklistContext = {
  payment: PaymentContext;
  identity: IdentityContext;
  event: EventContext;
  productCount: number;
};

export type BoothChecklistItem = {
  id:
    | "products"
    | "payment"
    | "name"
    | "description"
    | "event-location"
    | "event-hours";
  label: string;
  complete: boolean;
};

const isNonEmpty = (value: string) => value.trim().length > 0;

export function evaluateBoothChecklist(
  context: BoothChecklistContext,
): BoothChecklistItem[] {
  return [
    {
      id: "products",
      label: "Publish at least one product",
      complete: context.productCount > 0,
    },
    {
      id: "payment",
      label: "Configure bank-transfer details",
      complete:
        isNonEmpty(context.payment.bankName) &&
        isNonEmpty(context.payment.accountName) &&
        isNonEmpty(context.payment.accountNumber),
    },
    {
      id: "name",
      label: "Add a booth name",
      complete: isNonEmpty(context.identity.name),
    },
    {
      id: "description",
      label: "Add a booth description",
      complete: isNonEmpty(context.identity.description),
    },
    {
      id: "event-location",
      label: "Provide event location",
      complete: isNonEmpty(context.event.location),
    },
    {
      id: "event-hours",
      label: "Provide event hours",
      complete: isNonEmpty(context.event.hours),
    },
  ];
}

export function isChecklistComplete(items: BoothChecklistItem[]): boolean {
  return items.every((item) => item.complete);
}
```
